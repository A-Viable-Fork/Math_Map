-- Polynomial.degree (line 47)
def degree (p : R[X]) : WithBot ℕ :=
  p.support.max

-- Polynomial.natDegree (line 51)
def natDegree (p : R[X]) : ℕ :=
  (degree p).unbotD 0

-- Polynomial.degree_zero (line 76)
theorem degree_zero : degree (0 : R[X]) = ⊥

-- Polynomial.degree_add_le (line 329)
theorem degree_add_le (p q : R[X]) : degree (p + q) ≤ max (degree p) (degree q)

-- Polynomial.degree_mul_le (line 397)
theorem degree_mul_le (p q : R[X]) : degree (p * q) ≤ degree p + degree q

-- Polynomial.leadingCoeff (line 55)
def leadingCoeff (p : R[X]) : R :=
  coeff p (natDegree p)

-- Polynomial.Monic (line 59)
def Monic (p : R[X]) :=
  leadingCoeff p = (1 : R)

-- Polynomial.leadingCoeff_zero (line 353)
theorem leadingCoeff_zero : leadingCoeff (0 : R[X]) = 0
